## Useful Links Feature Setup

Run the following SQL commands in your Supabase SQL Editor to create the `useful_links` table and configure RLS permissions:

```sql
-- Create useful_links table
CREATE TABLE public.useful_links (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  link_name TEXT NOT NULL,
  link_description TEXT,
  link TEXT NOT NULL,
  roles INTEGER NOT NULL DEFAULT 1, -- Bitwise sum: 1=Guest, 2=Parent, 4=Staff, 8=Teacher, 16=Management, 32=Admin
  target TEXT NOT NULL DEFAULT '_blank', -- '_blank' (new window) or '_self' (current window)
  created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- Enable Row Level Security
ALTER TABLE public.useful_links ENABLE ROW LEVEL SECURITY;

-- Allow select access to everyone (filtering is handled on the client side based on role checks)
CREATE POLICY "Allow select on useful_links" 
ON public.useful_links 
FOR SELECT 
USING (true);

-- Allow all write operations to admin users (role & 32 != 0)
CREATE POLICY "Allow admin all on useful_links" 
ON public.useful_links 
ALL 
TO authenticated 
USING (
  EXISTS (
    SELECT 1 FROM public.user_roles 
    WHERE user_id = auth.uid() 
      AND (role & 32) != 0
  )
)
WITH CHECK (
  EXISTS (
    SELECT 1 FROM public.user_roles 
    WHERE user_id = auth.uid() 
      AND (role & 32) != 0
  )
);
```
